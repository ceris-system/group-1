/**
 * GET: Handles Data Fetching & Status Updates
 */
function doGet(e) {
  const sheetID = e.parameter.sheetID;
  const moduleName = e.parameter.module ? e.parameter.module.toLowerCase() : "";
  const action = e.parameter.action ? e.parameter.action.toLowerCase() : "";
  
  if (!sheetID) return msg({ error: "No Bridge ID provided" });

  try {
    const clientSS = SpreadsheetApp.openById(sheetID);

    // --- PLANTILLA DEACTIVATION LOOKUP ---
    if (moduleName === "deactivation" && (action === "lookup" || action === "list")) {
      const emploc = String(e.parameter.emploc || "").trim();
      if (action === "lookup" && !emploc) return msg({ error: "EMPLOC is required" });

      const masterSS = SpreadsheetApp.openById("15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg");
      const userDB = masterSS.getSheetByName("UserDB");
      if (!userDB) return msg({ error: "UserDB not found" });

      const userRows = userDB.getRange(2, 1, Math.max(userDB.getLastRow() - 1, 0), 13).getValues();
      let mikaSpreadsheetID = "";
      let clientName = "";
      let clientSheetName = "";
      for (let i = 0; i < userRows.length; i++) {
        if (String(userRows[i][5] || "").trim() === String(sheetID).trim()) {
          clientName = String(userRows[i][4] || "").trim();
          clientSheetName = String(userRows[i][3] || "").trim();
          mikaSpreadsheetID = String(userRows[i][11] || "").trim(); // UserDB column L
          break;
        }
      }
      if (!mikaSpreadsheetID) return msg({ error: "No lookup spreadsheet is mapped in UserDB column L" });

      const mikaSS = SpreadsheetApp.openById(mikaSpreadsheetID);
      const normalizedClientName = clientName.toUpperCase();
      const foreverDiserClients = ["GMA NORTH", "NCL", "GMA SOUTH", "SOUTH LUZON"];
      const targetSheetName = action === "list"
        ? (foreverDiserClients.some(name => normalizedClientName.includes(name)) ? "FOREVER DISER" : clientSheetName)
        : "mika file";
      const mikaSheet = mikaSS.getSheets().find(sheet => sheet.getName().trim().toLowerCase() === targetSheetName.toLowerCase());
      if (!mikaSheet) return msg({ error: `Sheet '${targetSheetName}' not found` });

      const lastRow = mikaSheet.getLastRow();
      if (action === "list") {
        const firstDataRow = 2;
        const dataRowCount = lastRow - firstDataRow + 1;
        if (dataRowCount <= 0) return msg({ success: true, rows: [] });
        return msg({ success: true, rows: mikaSheet.getRange(firstDataRow, 1, dataRowCount, 4).getDisplayValues() });
      }

      if (lastRow < 1) return msg({ error: "No records found in mika file" });

      const rows = mikaSheet.getRange(1, 1, lastRow, 2).getDisplayValues();
      const normalizedEmploc = emploc.toUpperCase();
      for (let i = 0; i < rows.length; i++) {
        if (String(rows[i][0] || "").trim().toUpperCase() === normalizedEmploc) {
          return msg({ success: true, emploc: emploc, fullName: rows[i][1] || "" });
        }
      }
      return msg({ success: false, error: "EMPLOC not found in mika file" });
    }

    // Helper for date formatting locked to Philippine time to avoid 1-day offset errors & formatted to Uppercase
    const fmtDate = (dateVal) => {
      if (!dateVal) return "---";
      if (typeof dateVal === "string" && dateVal.includes(",")) return dateVal.toUpperCase();
      
      const d = new Date(dateVal);
      if (isNaN(d.getTime())) return "---";
      
      // Explicitly lock to Philippine time zone and convert to Uppercase
      return Utilities.formatDate(d, "Asia/Manila", "MMMM dd, yyyy").toUpperCase();
    };

    // --- ROBUST MASTER MAPPING (EXPLICIT COLUMN LOOKUP) ---
    const getMasterMapping = () => {
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      if (!userDB) return { targetSpreadsheetID: sheetID, clientHeader: "" };

      const lastRow = userDB.getLastRow();
      if (lastRow < 2) return { targetSpreadsheetID: sheetID, clientHeader: "" };

      const userDbData = userDB.getRange(2, 1, lastRow - 1, 10).getValues();
      
      for (let i = 0; i < userDbData.length; i++) {
        const row = userDbData[i];
        const clientHeader = String(row[4] || "").trim(); // Column E (Index 4)
        const bridgeSheetID = String(row[5] || "").trim(); // Column F (Index 5)
        const targetSheetID = String(row[6] || "").trim(); // Column G (Index 6)

        if (bridgeSheetID === String(sheetID).trim()) {
          return {
            targetSpreadsheetID: targetSheetID || sheetID,
            clientHeader: clientHeader
          };
        }
      }
      return { targetSpreadsheetID: sheetID, clientHeader: "" };
    };

    // --- DEPLOYER LIST HANDLER FOR DROPDOWNS ---
    if (moduleName === "deployer_list") {
      const deployerSheet = clientSS.getSheetByName("DEPLOYER") || clientSS.getSheetByName("Deployer");
      if (!deployerSheet) return msg({ deployers: [], coordinators: [] });

      const lastRowD = deployerSheet.getLastRow();
      if (lastRowD < 2) return msg({ deployers: [], coordinators: [] });

      const deployerData = deployerSheet.getRange(2, 1, lastRowD - 1, 7).getValues();
      let deployers = [];
      let coordinators = [];

      for (let i = 0; i < deployerData.length; i++) {
        const dep = String(deployerData[i][0] || "").trim(); // Col A
        const coord = String(deployerData[i][6] || "").trim(); // Col G (Index 6)
        if (dep && !deployers.includes(dep)) deployers.push(dep);
        if (coord && !coordinators.includes(coord)) coordinators.push(coord);
      }
      return msg({ deployers: deployers, coordinators: coordinators });
    }

    // --- 2. VCODE MASTERLIST MODULE ---
    if (moduleName === "vcode masterlist") {
      const vcodeSheet = clientSS.getSheetByName("VCODE");
      if (!vcodeSheet) return msg({ error: "Sheet 'VCODE' not found" });

      const startRowV = 3;
      const lastRowV = vcodeSheet.getLastRow();
      if (lastRowV < startRowV) return msg([]);
      
      const data = vcodeSheet.getRange(startRowV, 1, lastRowV - (startRowV - 1), 14).getValues();
      return msg(data);
    }

    // --- 3. PLANTILLA MODULE ---
    if (moduleName === "plantilla") {
      const targetSheet = clientSS.getSheetByName("PLANTILLA") || clientSS.getSheets()[0];
      const startRowP = 9;
      const lastRowP = targetSheet.getLastRow();

      if (action === "marknewlyhired") {
        if (lastRowP < startRowP) return msg({ status: "success", message: "No data found" });

        const rangeAH = targetSheet.getRange(startRowP, 34, lastRowP - (startRowP - 1), 1);
        const valuesAH = rangeAH.getValues();
        let count = 0;

        for (let i = 0; i < valuesAH.length; i++) {
          if (!valuesAH[i][0] || String(valuesAH[i][0]).trim() === "") {
            valuesAH[i][0] = "NEWLY HIRED";
            count++;
          }
        }

        if (count > 0) {
          rangeAH.setValues(valuesAH);
        }
        return msg({ status: "success", message: `Marked ${count} empty records as NEWLY HIRED` });
      }

      if (action === "updaterecord") {
        const vcode = e.parameter.vcode; 
        const newRate = e.parameter.rate;          
        const newStatusAH = e.parameter.statusAH; 
        const newDate = e.parameter.updateDate;   

        if (!vcode) return msg({ status: "error", message: "VCODE missing" });
        const vcodes = targetSheet.getRange(startRowP, 2, lastRowP - (startRowP - 1), 1).getValues();
        
        for (let i = 0; i < vcodes.length; i++) {
          if (String(vcodes[i][0]) === String(vcode)) {
            const actualRow = startRowP + i;
            if (newRate !== undefined && newRate !== "") targetSheet.getRange(actualRow, 8).setValue(newRate);    
            if (newDate !== undefined && newDate !== "") targetSheet.getRange(actualRow, 33).setValue(newDate);    
            if (newStatusAH !== undefined && newStatusAH !== "") targetSheet.getRange(actualRow, 34).setValue(newStatusAH); 
            return msg({ status: "success", message: "Record Updated Successfully for VCODE: " + vcode });
          }
        }
        return msg({ status: "error", message: "VCODE not found" });
      }

      if (lastRowP < startRowP) return msg([]);
      const rawData = targetSheet.getRange(startRowP, 1, lastRowP - (startRowP - 1), 43).getValues();

      return msg(rawData.map(row => ({
          vcode: row[1] || "", 
          client: row[42] || "", 
          emploc: row[2] || "", 
          name: row[35] || "",
          position: row[6] || "", 
          rate: row[7] || "0", 
          area: row[8] || "",
          outlet: row[9] || "", 
          rest_day: row[10] || "", 
          status_m: row[12] || "",
          penalty: row[13] || "0", 
          date_hired: fmtDate(row[14]), 
          tenure: row[15] || "",
          sss: row[16] || "---", 
          philhealth: row[17] || "---", 
          pagibig: row[18] || "---",
          atm_no: row[19] || "---", 
          contact: row[20] || "---", 
          birthday: fmtDate(row[21]),
          age: row[22] || "---", 
          civil_status: row[23] || "", 
          address: row[24] || "",
          coordinator: row[25] || "", 
          hc: row[28] || 0, 
          date_ag: fmtDate(row[32]), 
          status: row[33] || "" 
      })));
    }

    // --- 4. ATTRITION REPORT MODULE ---
    if (moduleName === "attrition") {
      const attritionSheet = clientSS.getSheetByName("IN");
      if (!attritionSheet) return msg({ error: "Sheet 'IN' not found" });

      const firstDataRow = 3;
      const lastAttritionRow = attritionSheet.getLastRow();
      const dataRowCount = lastAttritionRow - firstDataRow + 1;
      if (dataRowCount <= 0) return msg({ rows: [] });

      const activeYear = new Date().getFullYear();
      const requestedMonth = String(e.parameter.month || "").trim().toUpperCase();
      const reportMode = String(e.parameter.mode || "monthly").trim().toLowerCase();
      const clientName = String(e.parameter.client || "").trim().toUpperCase();
      const monthNames = ["JANUARY", "FEBRUARY", "MARCH", "APRIL", "MAY", "JUNE", "JULY", "AUGUST", "SEPTEMBER", "OCTOBER", "NOVEMBER", "DECEMBER"];
      const normalizedMonth = requestedMonth || monthNames[new Date().getMonth()];
      const sourceColumnCount = Math.max(attritionSheet.getLastColumn(), 47); // AU contains the client filter.
      const sourceRange = attritionSheet.getRange(firstDataRow, 1, dataRowCount, sourceColumnCount);
      const sourceRows = sourceRange.getDisplayValues();
      const normalizeText = value => String(value || "").trim().replace(/\s+/g, " ").toUpperCase();
      const normalizeMonth = value => {
        const text = normalizeText(value);
        const numericMonth = Number(text);
        if (Number.isInteger(numericMonth) && numericMonth >= 1 && numericMonth <= 12) return monthNames[numericMonth - 1];
        return text.replace(/[^A-Z]/g, "");
      };
      const normalizeYear = value => {
        const match = String(value || "").match(/\d{4}/);
        return match ? match[0] : "";
      };
      const formatAttritionDate = value => {
        if (!value) return "";
        const dateValue = value instanceof Date ? value : new Date(value);
        if (isNaN(dateValue.getTime())) return String(value);
        return Utilities.formatDate(dateValue, "Asia/Manila", "MMMM dd, yyyy");
      };
      const normalizedClient = normalizeText(clientName);
      const clientAliases = normalizedClient
        .split(/[\\/,|]+/)
        .map(alias => alias.trim())
        .filter(alias => alias.length >= 3);
      const clientMatchesRow = rowClient => !normalizedClient || clientAliases.some(alias =>
        rowClient === alias || rowClient.includes(alias) || alias.includes(rowClient)
      );
      const normalizedRequestedMonth = normalizeMonth(normalizedMonth);
      let clientMatchCount = 0;
      let yearMatchCount = 0;
      let monthMatchCount = 0;
      const filteredRows = sourceRows.filter(row => {
        const rowClient = normalizeText(row[46]); // AU
        const rowMonth = normalizeMonth(row[42]); // AQ
        const rowYear = normalizeYear(row[43]); // AR
        const rowStatus = normalizeText(row[33]); // AH
        if (rowStatus.includes("MOVEMENT") || rowStatus.includes("RESHUFFLE")) return false;
        const clientMatches = clientMatchesRow(rowClient);
        const yearMatches = rowYear === String(activeYear);
        const monthMatches = reportMode !== "monthly" || rowMonth === normalizedRequestedMonth;
        if (clientMatches) clientMatchCount++;
        if (yearMatches) yearMatchCount++;
        if (monthMatches) monthMatchCount++;
        return clientMatches && yearMatches && monthMatches;
      });

      return msg({
        rows: filteredRows.map(row => [
          row[2] || "",                    // C: EMPLOC
          row[35] || "",                   // AJ: FULLNAME
          row[9] || "",                    // J: OUTLET
          row[8] || "",                    // I: AREA
          row[15] || "",                   // P: TENURE
          row[33] || "",                   // AH: STATUS
          row[20] || "",                   // U: CONTACT NO.
          row[24] || "",                   // Y: ADDRESS
          formatAttritionDate(row[14]),     // O: DATE HIRED
          formatAttritionDate(row[32])      // AG: DATE SEPARATION
        ]),
        activeYear: activeYear,
        diagnostics: {
          sourceRows: sourceRows.length,
          clientMatches: clientMatchCount,
          yearMatches: yearMatchCount,
          monthMatches: monthMatchCount,
          requestedClient: normalizedClient,
          requestedMonth: normalizedRequestedMonth
        }
      });
    }

    // --- 4. 5% BUFFER MODULE SPECIFIC HANDLER ---
    if (moduleName === "5_buffer") {
      try {
        const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
        const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
        const userDB = masterSS.getSheetByName("UserDB");
        
        if (!userDB) return msg({ error: "UserDB not found in Master Sheet." });

        let targetBufferSpreadsheetID = "";
        const lastRow = userDB.getLastRow();
        if (lastRow >= 2) {
          targetBufferSpreadsheetID = String(userDB.getRange(2, 7).getValue()).trim();
        }

        if (!targetBufferSpreadsheetID) {
          targetBufferSpreadsheetID = sheetID;
        }

        const sourceSS = SpreadsheetApp.openById(targetBufferSpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("-+5% GAP");
        
        if (!targetSheet) return msg({ error: "Sheet '-+5% GAP' not found in target spreadsheet." });
        
        const startRow = 8;
        const endRow = 21;
        const numRows = (endRow - startRow) + 1;

        const rangeValues = targetSheet.getRange(startRow, 3, numRows, 18).getValues();

        return msg({
          rows: rangeValues.map(row => ({
            account: row[0] || "",                            
            client_required: Math.round(row[1]) || 0,                
            on_board: Math.round(row[2]) || 0,                        
            yes_plantilla_not_mika: Math.round(row[5]) || 0,          
            yes_mika_not_plantilla: Math.round(row[6]) || 0,          
            yes_plantilla_not_payroll: Math.round(row[7]) || 0,       
            yes_payroll_not_plantilla: Math.round(row[8]) || 0,       
            payroll_buffer_1: row[11], 
            payroll_buffer_2: row[12], 
            mika_buffer_1: row[13], 
            mika_buffer_2: row[14], 
            commando_reliever: Math.round(row[15]) || 0,              
            reliever: Math.round(row[16]) || 0,                       
            total_active: Math.round(row[17]) || 0                    
          }))
        });

      } catch (err) {
        return msg({ error: "Buffer Error: " + err.toString() });
      }
    }

    // --- VACANCY MONITORING MODULE HANDLER ---
    if (moduleName === "vacancy_monitoring") {
      try {
        const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
        const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
        const userDB = masterSS.getSheetByName("UserDB");
        
        let vacancySpreadsheetID = sheetID;

        if (userDB) {
          const lastRow = userDB.getLastRow();
          if (lastRow >= 2) {
            const userDbData = userDB.getRange(2, 1, lastRow - 1, 10).getValues();
            for (let i = 0; i < userDbData.length; i++) {
              const row = userDbData[i];
              const bridgeSheetID = String(row[5] || "").trim(); 
              if (bridgeSheetID === String(sheetID).trim()) {
                vacancySpreadsheetID = String(row[8] || "").trim() || sheetID; 
                break;
              }
            }
          }
        }

        const sourceSS = SpreadsheetApp.openById(vacancySpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("VACANCY");
        
        if (!targetSheet) return msg({ error: "Sheet 'VACANCY' not found in spreadsheet ID: " + vacancySpreadsheetID });

        const startRow = 5; 
        const lastDataRow = targetSheet.getLastRow();
        if (lastDataRow < startRow) return msg({ rows: [] });

        const numRows = lastDataRow - (startRow - 1);
        const rawData = targetSheet.getRange(startRow, 1, numRows, targetSheet.getLastColumn()).getValues();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const vcode = String(row[1] || "").trim(); 

          const isRowEmpty = row.every(cell => String(cell).trim() === "");
          if (!isRowEmpty && vcode !== "") {
            row[15] = fmtDate(row[15]); 
            filteredRows.push(row); 
          }
        }

        return msg({ rows: filteredRows });

      } catch (err) {
        return msg({ error: "Vacancy Monitoring Error: " + err.toString() });
      }
    }
    
    // --- FOR APPROVAL MODULE HANDLER (GET) ---
    if (moduleName === "for_approval" || moduleName === "approval") {
      try {
        const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
        const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
        const userDB = masterSS.getSheetByName("UserDB");
        
        let approvalSpreadsheetID = sheetID;
        let approvalClientHeader = "";

        if (userDB) {
          const lastRow = userDB.getLastRow();
          if (lastRow >= 2) {
            const userDbData = userDB.getRange(2, 1, lastRow - 1, 13).getValues();
            for (let i = 0; i < userDbData.length; i++) {
              const row = userDbData[i];
              const bridgeSheetID = String(row[5] || "").trim(); 
              if (bridgeSheetID === String(sheetID).trim()) {
                approvalSpreadsheetID = String(row[11] || "").trim() || sheetID; 
                approvalClientHeader = String(row[4] || "").trim();
                break;
              }
            }
          }
        }

        const sourceSS = SpreadsheetApp.openById(approvalSpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("VACANCIES MONITORING") || sourceSS.getSheetByName("VACANCY") || sourceSS.getSheetByName("FOR APPROVAL");
        
        if (!targetSheet) return msg({ error: "Target sheet not found" });

        const startRow = 5; 
        const lastDataRow = targetSheet.getLastRow();
        if (lastDataRow < startRow) return msg({ rows: [] });

        const numRows = lastDataRow - (startRow - 1);
        
        const rawData = targetSheet.getRange(5, 1, numRows, targetSheet.getLastColumn()).getDisplayValues();
        let filteredRows = [];
        const normalizeClient = value => String(value || "")
          .replace(/\b(SYSTEM|SYSTM)\b/gi, "")
          .replace(/\s+/g, " ")
          .trim()
          .toUpperCase();
        const normalizedClient = normalizeClient(approvalClientHeader || e.parameter.clientHeader || "");
        const isAdmin = normalizedClient.includes("ADMIN");

        for (let i = 0; i < rawData.length; i++) {
          const row = rawData[i];

          const vcode = String(row[1] || "").trim();
          if (!vcode || vcode === "" || vcode.startsWith("=")) continue;

          const checkNotEmpty = (val) => {
            const str = String(val || "").trim();
            return str !== "" && str !== "---" && str !== "N/A" && !str.startsWith("=");
          };

          const hasAllFields = 
            checkNotEmpty(row[10]) && 
            checkNotEmpty(row[11]) && 
            checkNotEmpty(row[12]) && 
            checkNotEmpty(row[13]) && 
            checkNotEmpty(row[15]) && 
            checkNotEmpty(row[20]) && 
            checkNotEmpty(row[22]) && 
            checkNotEmpty(row[31]);   

          if (!hasAllFields) continue;

          const rowClient = normalizeClient(row[2]);
          if (!isAdmin && (!normalizedClient || rowClient !== normalizedClient)) continue;

          filteredRows.push({
            vcode: vcode,      
            colC: row[2] || "",  
            colB: vcode,        
            colG: row[6] || "",  
            colAL: row[37] || "",
            colI: row[8] || "",  
            colO: row[14] || "", 
            colP: row[15] || "", 
            colR: row[17] || ""  
          });
        }

        return msg({
          rows: filteredRows,
          clientHeader: approvalClientHeader,
          isAdmin: isAdmin
        });

      } catch (err) {
        return msg({ error: "For Approval Error: " + err.toString() });
      }
    }

    // --- 6. MIKA VS PLANTILLA DRILLDOWN MODULE ---
    if (moduleName === "mika_plantilla_drilldown") {
      try {
        const mapping = getMasterMapping();
        const sourceSS = SpreadsheetApp.openById(mapping.targetSpreadsheetID);
        const mikaSheet = sourceSS.getSheetByName("MIKA VS PLANTILLA1");
        
        if (!mikaSheet) return msg({ error: "Sheet 'MIKA VS PLANTILLA1' not found." });

        const lastDataRow = mikaSheet.getLastRow();
        if (lastDataRow < 7) return msg({ rows: [] });

        const rawData = mikaSheet.getRange(7, 1, lastDataRow - 6, 19).getValues();
        const lowerClientHeader = mapping.clientHeader.toLowerCase();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          if (mapping.clientHeader !== "" && String(row[18] || "").trim().toLowerCase() === lowerClientHeader) {
            filteredRows.push({
              EMPLOC: row[15] || "",          
              "MEMBERS NAME": row[16] || ""   
            });
          }
        }

        return msg({ rows: filteredRows });
      } catch (err) {
        return msg({ error: "Drilldown Error: " + err.toString() });
      }
    }

    // --- 7. MIKA NOT IN PLANTILLA DRILLDOWN MODULE ---
    if (moduleName === "mika_not_plantilla_drilldown") {
      try {
        const mapping = getMasterMapping();
        const sourceSS = SpreadsheetApp.openById(mapping.targetSpreadsheetID);
        const mikaSheet = sourceSS.getSheetByName("MIKA VS PLANTILLA1");
        
        if (!mikaSheet) return msg({ error: "Sheet 'MIKA VS PLANTILLA1' not found." });

        const lastDataRow = mikaSheet.getLastRow();
        if (lastDataRow < 7) return msg({ rows: [] });

        const rawData = mikaSheet.getRange(7, 7, lastDataRow - 6, 4).getValues();
        const lowerClientHeader = mapping.clientHeader.toLowerCase();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const colJValue = String(row[3] || "").trim();

          if (colJValue !== "" && (mapping.clientHeader === "" || colJValue.toLowerCase() === lowerClientHeader)) {
            filteredRows.push({
              EMPLOC: row[0] || "",          
              "MEMBERS NAME": row[1] || ""   
            });
          }
        }

        return msg({ rows: filteredRows });
      } catch (err) {
        return msg({ error: "Drilldown Error: " + err.toString() });
      }
    }

    // --- 8. PAYROLL VS PLANTILLA DRILLDOWN (TAMAD) MODULE ---
    if (moduleName === "payroll_plantilla_tamad_drilldown") {
      try {
        const mapping = getMasterMapping();
        const sourceSS = SpreadsheetApp.openById(mapping.targetSpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("PAYROLL VS PLANTILLA1");
        
        if (!targetSheet) return msg({ error: "Sheet 'PAYROLL VS PLANTILLA1' not found." });

        const lastDataRow = targetSheet.getLastRow();
        if (lastDataRow < 7) return msg({ rows: [] });

        const rawData = targetSheet.getRange(7, 18, lastDataRow - 6, 5).getValues();
        const lowerClientHeader = mapping.clientHeader.toLowerCase();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const colVValue = String(row[4] || "").trim();

          if (colVValue !== "" && (mapping.clientHeader === "" || colVValue.toLowerCase() === lowerClientHeader)) {
            filteredRows.push({
              EMPLOC: row[0] || "",          
              "MEMBERS NAME": row[1] || ""   
            });
          }
        }

        return msg({ rows: filteredRows });

      } catch (err) {
        return msg({ error: "Drilldown Error: " + err.toString() });
      }
    }

    // --- 9. PAYROLL VS PLANTILLA GHOST DRILLDOWN MODULE ---
    if (moduleName === "payroll_plantilla_ghost_drilldown") {
      try {
        const mapping = getMasterMapping();
        const sourceSS = SpreadsheetApp.openById(mapping.targetSpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("PAYROLL VS PLANTILLA1");
        
        if (!targetSheet) return msg({ error: "Sheet 'PAYROLL VS PLANTILLA1' not found." });

        const lastDataRow = targetSheet.getLastRow();
        if (lastDataRow < 7) return msg({ rows: [] });

        const rawData = targetSheet.getRange(7, 18, lastDataRow - 6, 5).getValues();
        const lowerClientHeader = mapping.clientHeader.toLowerCase();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const colVValue = String(row[4] || "").trim();

          if (colVValue !== "" && (mapping.clientHeader === "" || colVValue.toLowerCase() === lowerClientHeader)) {
            filteredRows.push({
              EMPLOC: row[0] || "",          
              "MEMBERS NAME": row[1] || ""   
            });
          }
        }

        return msg({ rows: filteredRows });

      } catch (err) {
        return msg({ error: "Drilldown Error: " + err.toString() });
      }
    }

    // --- VCODE VARIANCE MODULE ---
    if (moduleName === "variance" || moduleName === "vcode_variance") {
      try {
        const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
        const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
        const userDB = masterSS.getSheetByName("UserDB");
        
        let varianceSpreadsheetID = sheetID;
        let clientHeader = "";

        if (userDB) {
          const lastRow = userDB.getLastRow();
          if (lastRow >= 2) {
            const userDbData = userDB.getRange(2, 1, lastRow - 1, 8).getValues();
            for (let i = 0; i < userDbData.length; i++) {
              const row = userDbData[i];
              const bridgeSheetID = String(row[5] || "").trim(); 
              if (bridgeSheetID === String(sheetID).trim()) {
                clientHeader = String(row[4] || "").trim();         
                varianceSpreadsheetID = String(row[7] || "").trim() || sheetID; 
                break;
              }
            }
          }
        }

        const sourceSS = SpreadsheetApp.openById(varianceSpreadsheetID);
        const targetSheet = sourceSS.getSheetByName("SUMMARY") || 
                            sourceSS.getSheetByName("Summary");
        
        if (!targetSheet) return msg({ error: "Sheet 'SUMMARY' not found in spreadsheet ID: " + varianceSpreadsheetID });

        const lastDataRow = targetSheet.getLastRow();
        if (lastDataRow < 3) return msg({ rows: [] });

        const rawData = targetSheet.getRange(3, 2, lastDataRow - 2, 5).getValues();
        const lowerClientHeader = clientHeader.toLowerCase();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const vcode = String(row[0] || "").trim();      
          const colFValue = String(row[4] || "").trim();  

          if (vcode !== "") {
            if (clientHeader === "" || colFValue.toLowerCase() === lowerClientHeader) {
              filteredRows.push({
                VCODE: row[0] || "",       
                OUTLET: row[1] || "",      
                POSITION: row[2] || "",    
                HC: row[3] || ""           
              });
            }
          }
        }

        return msg({ rows: filteredRows });

      } catch (err) {
        return msg({ error: "Vcode Variance Error: " + err.toString() });
      }
    }

    // --- HR EMPLOC MONITORING MODULE HANDLER ---
    if (moduleName === "hr_emploc_monitoring") {
      try {
        const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
        const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
        const userDB = masterSS.getSheetByName("UserDB");
        
        let emplocSpreadsheetID = sheetID;
        let userClientName = "";

        if (userDB) {
          const lastRow = userDB.getLastRow();
          if (lastRow >= 2) {
            const userDbData = userDB.getRange(2, 1, lastRow - 1, 10).getValues();
            for (let i = 0; i < userDbData.length; i++) {
              const row = userDbData[i];
              const bridgeSheetID = String(row[5] || "").trim(); 
              if (bridgeSheetID === String(sheetID).trim()) {
                emplocSpreadsheetID = String(row[9] || "").trim() || sheetID; 
                userClientName = String(row[4] || "").trim(); 
                break;
              }
            }
          }
        }

        const sourceSS = SpreadsheetApp.openById(emplocSpreadsheetID);
        const sheets = sourceSS.getSheets();
        let targetSheet = null;
        for (let s = 0; s < sheets.length; s++) {
          const sName = sheets[s].getName().trim().toUpperCase();
          if (sName === "G1N") {
            targetSheet = sheets[s];
            break;
          }
        }
        
        if (!targetSheet) return msg({ error: "Sheet 'G1N' not found in spreadsheet ID: " + emplocSpreadsheetID });

        const lastDataRow = targetSheet.getLastRow();
        const startRow = 9; 
        if (lastDataRow < startRow) return msg({ rows: [], clientHeader: userClientName });

        const numRows = lastDataRow - (startRow - 1);
        const rawData = targetSheet.getRange(startRow, 1, numRows, targetSheet.getLastColumn()).getValues();
        let filteredRows = [];

        for (let i = 0, len = rawData.length; i < len; i++) {
          const row = rawData[i];
          const vcode = String(row[6] || "").trim(); 

          if (vcode !== "") {
            filteredRows.push(row);
          }
        }

        return msg({ rows: filteredRows, clientHeader: userClientName });

      } catch (err) {
        return msg({ error: "HR Emploc Monitoring Error: " + err.toString() });
      }
    }

    // --- 5. GENERIC MAPPING ---
    const sheetMapping = {
      "plantilla_summary": "SUMMARY",
      "sil_system": "SIL", 
      "hr_emploc_monitoring": "EMPLOC",
    };

    if (sheetMapping[moduleName]) {
      const sheet = clientSS.getSheetByName(sheetMapping[moduleName]);
      if (!sheet) return msg({ error: `Sheet '${sheetMapping[moduleName]}' not found` });
      return msg(sheet.getDataRange().getValues());
    }

    return msg(clientSS.getSheets()[0].getDataRange().getValues());

  } catch (err) {
    return msg({ error: "System Error: " + err.message });
  }
}

/**
 * UNIFIED POST ROUTER: Handles Authentication, Approvals, Updates, Transfers, and Deletions
 */
function doPost(e) {
  try {
    const rawContent = e.postData ? e.postData.contents : "";
    const data = rawContent ? JSON.parse(rawContent) : {};
    const sheetID = e.parameter.sheetID;
    
    const moduleName = (data.module || e.parameter.module || "").toLowerCase();
    const action = (data.action || e.parameter.action || "").toLowerCase();

    // --- PLANTILLA DEACTIVATION SAVE ---
    if (moduleName === "deactivation" && action === "save") {
      if (!sheetID) return msg({ status: "error", message: "Bridge ID provided is missing" });

      const emploc = String(data.emploc || "").trim();
      const fullName = String(data.fullName || "").trim();
      const status = String(data.status || "").trim();
      const inactiveDate = String(data.inactiveDate || "").trim();
      if (!emploc || !fullName || !status || !inactiveDate) {
        return msg({ status: "error", message: "EMPLOC, full name, status, and inactive date are required" });
      }

      const masterSS = SpreadsheetApp.openById("15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg");
      const userDB = masterSS.getSheetByName("UserDB");
      if (!userDB) return msg({ status: "error", message: "UserDB not found" });

      const rows = userDB.getRange(2, 1, Math.max(userDB.getLastRow() - 1, 0), 13).getValues();
      let targetSpreadsheetID = "";
      let clientSheetName = "";
      let clientName = "";
      for (let i = 0; i < rows.length; i++) {
        if (String(rows[i][5] || "").trim() === String(sheetID).trim()) {
          clientName = String(rows[i][4] || "").trim(); // Column E: client name
          clientSheetName = String(rows[i][3] || "").trim(); // Column D
          targetSpreadsheetID = String(rows[i][11] || "").trim(); // Column L
          break;
        }
      }
      if (!targetSpreadsheetID) return msg({ status: "error", message: "No deactivation spreadsheet is mapped in UserDB column L" });

      const targetSS = SpreadsheetApp.openById(targetSpreadsheetID);
      const normalizedClientName = clientName.toUpperCase();
      const foreverDiserClients = ["GMA NORTH", "NCL", "GMA SOUTH", "SOUTH LUZON"];
      const useForeverDiserSheet = foreverDiserClients.some(name => normalizedClientName.includes(name));
      const targetSheetName = useForeverDiserSheet ? "FOREVER DISER" : clientSheetName;
      const targetSheet = targetSS.getSheetByName(targetSheetName)
        || targetSS.getSheetByName("DEACTIVATION")
        || targetSS.getSheetByName("Deactivation");
      if (!targetSheet) return msg({ status: "error", message: "Client deactivation sheet not found" });

      targetSheet.appendRow([emploc, fullName, status, inactiveDate]);
      return msg({ status: "success", message: "Saved to deactivation records" });
    }

    // --- 1. VCODE VARIANCE: TRANSFER ACTION ---
    if (moduleName === "vcode_variance" && action === "transfer") {
      var vcodes = data.vcodes;
      var targetSSId = data.targetSpreadsheetID;
      var targetSheetName = data.targetSheet || "VACANCY";
      
      var targetSS = SpreadsheetApp.openById(targetSSId);
      var vacancySheet = targetSS.getSheetByName(targetSheetName);
      
      if (!vacancySheet) return msg({ success: false, error: "Target sheet 'VACANCY' not found" });

      vcodes.forEach(function(vcode) {
        var lastRow = vacancySheet.getLastRow() + 1;
        vacancySheet.getRange(lastRow, 2).setValue(vcode); 
      });
      
      return msg({ success: true, message: "Transferred successfully to VACANCY." });
    }

    // --- 2. VCODE VARIANCE: DELETE ACTION ---
    if (moduleName === "vcode_variance" && action === "delete") {
      var vcodes = data.vcodes;
      
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      
      var targetSSId = "";
      if (userDB) {
        const lastRowDB = userDB.getLastRow();
        if (lastRowDB >= 2) {
          const userDbData = userDB.getRange(2, 1, lastRowDB - 1, 10).getValues();
          for (let i = 0; i < userDbData.length; i++) {
            const row = userDbData[i];
            const bridgeSheetID = String(row[5] || "").trim(); 
            if (bridgeSheetID === String(sheetID).trim()) {
              targetSSId = String(row[5] || "").trim(); 
              break;
            }
          }
        }
      }

      if (!targetSSId) {
        targetSSId = sheetID;
      }

      var targetSS = SpreadsheetApp.openById(targetSSId);
      var deletedSheet = targetSS.getSheetByName("DELETED VCODES");
      var vcodeSourceSheet = targetSS.getSheetByName("VCODE"); 
      
      if (!deletedSheet || !vcodeSourceSheet) return msg({ success: false, error: "Required archive sheets missing ('DELETED VCODES' or 'VCODE')" });

      vcodes.forEach(function(targetVcode) {
        var vcodeData = vcodeSourceSheet.getDataRange().getValues();
        for (let i = vcodeData.length - 1; i >= 1; i--) {
          if (vcodeData[i].some(cell => String(cell).trim() === String(targetVcode).trim())) {
            var entireRow = vcodeData[i];
            var lastRowDeleted = deletedSheet.getLastRow() + 1;
            deletedSheet.getRange(lastRowDeleted, 1, 1, entireRow.length).setValues([entireRow]);
            vcodeSourceSheet.deleteRow(i + 1);
            break;
          }
        }
      });

      return msg({ success: true, message: "Archived to DELETED VCODES and removed." });
    }

    // --- 3. APPROVE AND TRANSFER MODULE ACTION ---
    if (moduleName === "for_approval" && (action === "approvetransfer" || action === "approve_transfer" || action === "approve")) {
      if (!sheetID) return msg({ status: "error", message: "Bridge ID provided is missing" });
      
      const vcodesToProcess = data.vcodes; 
      if (!vcodesToProcess || !Array.isArray(vcodesToProcess) || vcodesToProcess.length === 0) {
        return msg({ status: "error", message: "No VCODES provided for approval transfer" });
      }

      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      
      let vacancySpreadsheetID = sheetID;
      if (userDB) {
        const lastRow = userDB.getLastRow();
        if (lastRow >= 2) {
          const userDbData = userDB.getRange(2, 1, lastRow - 1, 13).getValues();
          for (let i = 0; i < userDbData.length; i++) {
            const row = userDbData[i];
            const bridgeSheetID = String(row[5] || "").trim();
            if (bridgeSheetID === String(sheetID).trim()) {
              vacancySpreadsheetID = String(row[11] || "").trim() || sheetID; 
              break;
            }
          }
        }
      }

      const sourceSS = SpreadsheetApp.openById(vacancySpreadsheetID);
      const vacancySheet = sourceSS.getSheetByName("VACANCIES MONITORING") || sourceSS.getSheetByName("VACANCY") || sourceSS.getSheetByName("FOR APPROVAL");
      if (!vacancySheet) return msg({ status: "error", message: "Source Vacancy sheet not found" });

      const hrEmplocSSID = "1DxY_U6XAH-03DzBoBpU7je_WCpoUYz8t6XmrQpWCL3s";
      const onBoardDBSSID = "1GRNJbY1xoLuhrCVOduSo9DBACseMHJa9s5l5W4rOc5Y";

      const hrEmplocSheet = SpreadsheetApp.openById(hrEmplocSSID).getSheetByName("G1N");
      const onBoardSS = SpreadsheetApp.openById(onBoardDBSSID);
      const onBoardSheet = onBoardSS.getSheetByName("On Board Database") || onBoardSS.getSheets()[0];

      if (!hrEmplocSheet) return msg({ status: "error", message: "Target HR Emploc sheet 'G1N' not found" });
      if (!onBoardSheet) return msg({ status: "error", message: "Target On Board Database sheet not found" });

      const lastDataRow = vacancySheet.getLastRow();
      if (lastDataRow < 5) return msg({ status: "error", message: "No records found in Vacancy sheet" });

      const numRows = lastDataRow - 4;
      const rawData = vacancySheet.getRange(5, 1, numRows, vacancySheet.getLastColumn()).getValues();

      let processedCount = 0;
      for (let i = rawData.length - 1; i >= 0; i--) {
        const row = rawData[i];
        const vcode = String(row[1] || "").trim(); 

        if (vcodesToProcess.includes(vcode)) {
          const actualRowIndex = 5 + i;
          const colZ = row[25] || "";
          const colC = row[2] || "";
          const colK = row[10] || "";
          const colL = row[11] || "";
          const colM = row[12] || "";
          const colV = row[21] || "";
          const colB = row[1] || "";

          hrEmplocSheet.appendRow([colZ, colC, colK, colL, colM, colV, colB]);
          onBoardSheet.appendRow(row);
          vacancySheet.deleteRow(actualRowIndex);

          processedCount++;
        }
      }

      return msg({ status: "success", message: `Successfully approved and transferred ${processedCount} record(s).` });
    }

    // --- 3.5. PLANTILLA SIDE DRAWER STATUS TRANSFER MODULE ---
    if (moduleName === "plantilla" && (action === "transfer" || action === "update_status" || action === "inactive" || action === "side_drawer_update")) {
      if (!sheetID) return msg({ status: "error", message: "Bridge ID provided is missing" });

      const vcodesToProcess = data.vcodes || [data.vcode];
      if (!vcodesToProcess || vcodesToProcess.length === 0 || !vcodesToProcess[0]) {
        return msg({ status: "error", message: "No VCODES provided for Plantilla update" });
      }

      const clientSS = SpreadsheetApp.openById(sheetID);
      const plantillaSheet = clientSS.getSheetByName("PLANTILLA") || clientSS.getSheets()[0];
      if (!plantillaSheet) return msg({ status: "error", message: "Plantilla sheet not found" });

      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      
      let vacancySSID = sheetID;
      let inactiveSSID = sheetID;
      let inactiveSheetName = "";

      if (userDB) {
        const lastRowDB = userDB.getLastRow();
        if (lastRowDB >= 2) {
          const userDbData = userDB.getRange(2, 1, lastRowDB - 1, 15).getValues();
          for (let i = 0; i < userDbData.length; i++) {
            const row = userDbData[i];
            const bridgeSheetID = String(row[5] || "").trim();
            if (bridgeSheetID === String(sheetID).trim()) {
              inactiveSheetName = String(row[3] || "").trim();  // Column D: designated Inactive sheet name
              vacancySSID = String(row[8] || "").trim() || sheetID;    // Column I (Index 8) for Vacancy spreadsheet
              inactiveSSID = String(row[10] || "").trim() || sheetID;  // Column K (Index 10) for Inactive spreadsheet
              break;
            }
          }
        }
      }

      const vacancySS = SpreadsheetApp.openById(vacancySSID);
      const vacancySheet = vacancySS.getSheetByName("VACANCY");
      if (!vacancySheet) return msg({ status: "error", message: "Target sheet 'VACANCY' not found" });

      const inactiveSS = SpreadsheetApp.openById(inactiveSSID);
      const normalizedInactiveName = inactiveSheetName.replace(/\s*(SYSTEM|SYSTM)\s*$/i, "").trim();
      const inactiveSheet = (normalizedInactiveName && inactiveSS.getSheetByName(normalizedInactiveName))
        || (inactiveSheetName && inactiveSS.getSheetByName(inactiveSheetName))
        || inactiveSS.getSheetByName("INACTIVE")
        || inactiveSS.getSheetByName("Inactive");
      if (!inactiveSheet) return msg({ status: "error", message: "Target 'INACTIVE' sheet not found" });

      const sheetContainsVcode = (sheet, vcode) => {
        const values = sheet.getDataRange().getDisplayValues();
        const target = String(vcode).trim();
        return values.some(row => row.some(cell => String(cell).trim() === target));
      };

      const startRowP = 9;
      const lastRowP = plantillaSheet.getLastRow();
      if (lastRowP < startRowP) return msg({ status: "error", message: "No records found in Plantilla" });

      const vcodeCol = plantillaSheet.getRange(startRowP, 2, lastRowP - (startRowP - 1), 1).getValues();
      let processedCount = 0;

      const validStatuses = ["RESIGNED", "AWOL", "TERMINATED", "MOVEMENT", "BACK OUT", "ENDO", "FLOATING"];

      for (let i = vcodeCol.length - 1; i >= 0; i--) {
        const currentVcode = String(vcodeCol[i][0]).trim();
        
        if (vcodesToProcess.includes(currentVcode)) {
          const actualRowIndex = startRowP + i;
          
          // Optional status and date checks if sent in payload
          const rowStatus = String(data.status || plantillaSheet.getRange(actualRowIndex, 34).getValue()).trim().toUpperCase();
          const rowDate = data.updateDate || plantillaSheet.getRange(actualRowIndex, 33).getValue();

          // Check if status matches one of the specified conditions and a date exists
          if (validStatuses.includes(rowStatus) && rowDate) {
            const entireRowData = plantillaSheet.getRange(actualRowIndex, 1, 1, plantillaSheet.getLastColumn()).getValues()[0];
            const inactiveRowData = entireRowData.slice(0, 38); // Plantilla columns A:AL

            // Append the complete A:AL row to the mapped Inactive tab.
            inactiveSheet.appendRow(inactiveRowData);

            if (!sheetContainsVcode(vacancySheet, currentVcode)) {
              const lastVacancyRow = vacancySheet.getLastRow() + 1;
              vacancySheet.getRange(lastVacancyRow, 2).setValue(currentVcode);
            }

            // Delete the source row only after both destinations are present.
            plantillaSheet.deleteRow(actualRowIndex);
            processedCount++;
          }
        }
      }

      return msg({ 
        status: "success", 
        message: `Successfully processed and transferred ${processedCount} record(s) based on status conditions.` 
      });
    }

    // --- 4. VACANCY RECORD UPDATE ACTION (STRICT VCODE LOOKUP & SELECTIVE UPDATE) ---
    if (moduleName === "vacancy_monitoring" && action === "updaterecord") {
      if (!sheetID) return msg({ status: "error", message: "Bridge ID provided is missing" });
      
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      
      let vacancySpreadsheetID = sheetID;
      if (userDB) {
        const lastRow = userDB.getLastRow();
        if (lastRow >= 2) {
          const userDbData = userDB.getRange(2, 1, lastRow - 1, 10).getValues();
          for (let i = 0; i < userDbData.length; i++) {
            const row = userDbData[i];
            const bridgeSheetID = String(row[5] || "").trim();
            if (bridgeSheetID === String(sheetID).trim()) {
              vacancySpreadsheetID = String(row[8] || "").trim() || sheetID; 
              break;
            }
          }
        }
      }

      const clientSS = SpreadsheetApp.openById(vacancySpreadsheetID);
      const vacancySheet = clientSS.getSheetByName("VACANCY");
      if (!vacancySheet) return msg({ status: "error", message: "Sheet 'VACANCY' not found" });

      const vcode = data.vcode || e.parameter.vcode;
      if (!vcode) return msg({ status: "error", message: "VCODE missing in payload" });

      const lastRow = vacancySheet.getLastRow();
      if (lastRow < 5) return msg({ status: "error", message: "No records found in VACANCY sheet" });

      const vcodeColumn = vacancySheet.getRange(5, 2, lastRow - 4, 1).getValues(); 
      let targetRowIndex = -1;

      for (let i = 0; i < vcodeColumn.length; i++) {
        if (String(vcodeColumn[i][0]).trim() === String(vcode).trim()) {
          targetRowIndex = 5 + i; 
          break;
        }
      }

      if (targetRowIndex === -1) return msg({ status: "error", message: "VCODE not found inside VACANCY sheet" });

      if (data.lastName !== undefined && data.lastName !== "") vacancySheet.getRange(targetRowIndex, 11).setValue(data.lastName);       
      if (data.firstName !== undefined && data.firstName !== "") vacancySheet.getRange(targetRowIndex, 12).setValue(data.firstName);     
      if (data.middleName !== undefined && data.middleName !== "") vacancySheet.getRange(targetRowIndex, 13).setValue(data.middleName);   
      if (data.contact !== undefined && data.contact !== "") vacancySheet.getRange(targetRowIndex, 14).setValue(data.contact);         
      if (data.onboardDate !== undefined && data.onboardDate !== "") vacancySheet.getRange(targetRowIndex, 16).setValue(data.onboardDate); 
      if (data.reliever !== undefined && data.reliever !== "") vacancySheet.getRange(targetRowIndex, 21).setValue(data.reliever);       
      if (data.remarks !== undefined && data.remarks !== "") vacancySheet.getRange(targetRowIndex, 23).setValue(data.remarks);         
      if (data.coordinator !== undefined && data.coordinator !== "") vacancySheet.getRange(targetRowIndex, 30).setValue(data.coordinator); 
      if (data.deployedBy !== undefined && data.deployedBy !== "") vacancySheet.getRange(targetRowIndex, 32).setValue(data.deployedBy);   

      return msg({ status: "success", message: "Record Updated Successfully for VCODE: " + vcode });
    }

    // --- 5. HR EMPLOC MONITORING RECORD UPDATE ACTION (STRICT VCODE LOOKUP) ---
    if (moduleName === "hr_emploc_monitoring" && action === "updaterecord") {
      if (!sheetID) return msg({ status: "error", message: "Bridge ID provided is missing" });
      
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const masterSS = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const userDB = masterSS.getSheetByName("UserDB");
      
      let emplocSpreadsheetID = sheetID;
      if (userDB) {
        const lastRow = userDB.getLastRow();
        if (lastRow >= 2) {
          const userDbData = userDB.getRange(2, 1, lastRow - 1, 20).getValues();
          for (let i = 0; i < userDbData.length; i++) {
            const row = userDbData[i];
            const bridgeSheetID = String(row[5] || "").trim();
            if (bridgeSheetID === String(sheetID).trim()) {
              emplocSpreadsheetID = String(row[9] || "").trim() || sheetID; 
              break;
            }
          }
        }
      }

      const clientSS = SpreadsheetApp.openById(emplocSpreadsheetID);
      const emplocSheet = clientSS.getSheetByName("G1N");
      if (!emplocSheet) return msg({ status: "error", message: "Sheet 'G1N' not found" });

      const vcode = data.vcode || e.parameter.vcode;
      if (!vcode) return msg({ status: "error", message: "VCODE missing in payload" });

      const lastRow = emplocSheet.getLastRow();
      if (lastRow < 9) return msg({ status: "error", message: "No records found in G1N sheet" });

      const vcodeColumn = emplocSheet.getRange(9, 7, lastRow - 8, 1).getValues(); 
      let targetRowIndex = -1;

      for (let i = 0; i < vcodeColumn.length; i++) {
        if (String(vcodeColumn[i][0]).trim() === String(vcode).trim()) {
          targetRowIndex = 9 + i; 
          break;
        }
      }

      if (targetRowIndex === -1) return msg({ status: "error", message: "VCODE not found inside G1N sheet" });
      if (data.remarks !== undefined && data.remarks !== "") emplocSheet.getRange(targetRowIndex, 10).setValue(data.remarks);

      return msg({ status: "success", message: "Record Updated Successfully for VCODE: " + vcode });
    }

    // --- 5.4. FORGOT PASSWORD: ACCOUNT LOOKUP ACTION ---
    if (action === "lookupaccountforreset") {
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg";
      const ss = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const sheet = ss.getSheetByName("UserDB");

      const lastRow = sheet.getLastRow();
      const rows = sheet.getRange(1, 1, lastRow, 20).getValues();

      const lookupUser = String(data.user || "").trim().toUpperCase();
      if (!lookupUser) return msg({ success: false, error: "Username is required." });

      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim().toUpperCase() === lookupUser) {
          return msg({
            success: true,
            rowIndex: i + 1,
            username: String(rows[i][0]).trim()
          });
        }
      }
      return msg({ success: false, error: "Username not found." });
    }

    // --- 5.5. UPDATE PASSWORD ACTION ---
    if (action === "updatepassword") {
      const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg"; 
      const ss = SpreadsheetApp.openById(MASTER_SHEET_ID);
      const sheet = ss.getSheetByName("UserDB"); 
      
      const lastRow = sheet.getLastRow();
      const rows = sheet.getRange(1, 1, lastRow, 20).getValues();
      
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim().toUpperCase() === String(data.user).trim().toUpperCase()) {
          const targetRow = i + 1;
          
          sheet.getRange(targetRow, 2).setValue(data.newPass);
          sheet.getRange(targetRow, 3).setValue("ACTIVE");
          
          let bridgesObj = {};
          const plantillaID    = String(rows[i][5] || "").trim();  
          const vacancyID      = String(rows[i][8] || "").trim();  
          const hrEmplocID     = String(rows[i][9] || "").trim();  
          const attritionID    = String(rows[i][10] || "").trim(); 
          const approvalID     = String(rows[i][11] || "").trim(); 
          const deactivationID = String(rows[i][12] || "").trim(); 

          if (plantillaID)    bridgesObj["plantilla"] = plantillaID;
          if (vacancyID)      bridgesObj["vacancy_monitoring"] = vacancyID;
          if (hrEmplocID)     bridgesObj["hr_emploc_monitoring"] = hrEmplocID;
          if (attritionID)    bridgesObj["attrition"] = attritionID;
          if (deactivationID) bridgesObj["deactivation"] = deactivationID;
          if (approvalID)     bridgesObj["approval"] = approvalID;

          return msg({
            success: true, 
            status: "ACTIVE",       
            clientName: rows[i][4],   
            bridges: bridgesObj
          });
        }
      }
      return msg({ success: false, error: "User not found for password update." });
    }

    // --- 6. STANDARD LOGIN / AUTHENTICATION FALLBACK ---
    const MASTER_SHEET_ID = "15R18IM11tQHdih52xuxn0pwT9YFPVZLT0ispX8uzjKg"; 
    const ss = SpreadsheetApp.openById(MASTER_SHEET_ID);
    const sheet = ss.getSheetByName("UserDB"); 
    
    const lastRow = sheet.getLastRow();
    const rows = sheet.getRange(1, 1, lastRow, 20).getValues();
    
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim().toUpperCase() === String(data.user).trim().toUpperCase() && String(rows[i][1]).trim() === String(data.pass).trim()) {
        let bridgesObj = {};
        const plantillaID    = String(rows[i][5] || "").trim();  
        const vacancyID      = String(rows[i][8] || "").trim();  
        const hrEmplocID     = String(rows[i][9] || "").trim();  
        const attritionID    = String(rows[i][10] || "").trim(); 
        const approvalID     = String(rows[i][11] || "").trim(); 
        const deactivationID = String(rows[i][12] || "").trim(); 

        if (plantillaID)    bridgesObj["plantilla"] = plantillaID;
        if (vacancyID)      bridgesObj["vacancy_monitoring"] = vacancyID;
        if (hrEmplocID)     bridgesObj["hr_emploc_monitoring"] = hrEmplocID;
        if (attritionID)    bridgesObj["attrition"] = attritionID;
        if (deactivationID) bridgesObj["deactivation"] = deactivationID;
        if (approvalID)     bridgesObj["approval"] = approvalID;

        return msg({
          success: true, 
          status: rows[i][2],       
          clientName: rows[i][4],   
          bridges: bridgesObj,
          rowIndex: i + 1
        });
      }
    }
    return msg({ success: false, error: "Invalid Credentials" });
    
  } catch (err) {
    return msg({ success: false, error: "Auth Error: " + err.message });
  }
}

/**
 * Helper to return JSON output securely without triggering browser fetch CORS block exceptions
 */
function msg(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
